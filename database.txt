import sqlite3
from pathlib import Path
import random
import string

DB_NAME = "truth_or_dare.db"

def get_connection():
    return sqlite3.connect(DB_NAME)

def init_db():
    conn = get_connection()
    cursor = conn.cursor()

    # USERS TABLE
    cursor.execute("""
    CREATE TABLE IF NOT EXISTS users (
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        telegram_id INTEGER UNIQUE NOT NULL,
        username TEXT
    )
    """)

    # PROMPTS TABLE
    cursor.execute("""
        CREATE TABLE IF NOT EXISTS prompts (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            type TEXT NOT NULL,
            difficulty TEXT NOT NULL,
            game_type TEXT NOT NULL,
            text TEXT NOT NULL
        )
    """)

    cursor.execute("""
    CREATE TABLE IF NOT EXISTS game_history (
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        game_code TEXT,
        challenged_player INTEGER,
        prompt_type TEXT,
        prompt_text TEXT,
        answer_text TEXT,
        created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
    )
    """)

    # GAMES TABLE
    cursor.execute("""
    CREATE TABLE IF NOT EXISTS games (
        code TEXT PRIMARY KEY,
        player1_id INTEGER NOT NULL,
        player2_id INTEGER,
        current_turn INTEGER,
        state TEXT DEFAULT 'WAITING',
        game_type TEXT,
        difficulty TEXT,
        current_prompt TEXT
    )
    """)

    conn.commit()
    conn.close()

def create_or_update_user(telegram_id: int, username: str):
    conn = get_connection()
    cursor = conn.cursor()

    cursor.execute("""
        INSERT INTO users (telegram_id, username)
        VALUES (?, ?)
        ON CONFLICT(telegram_id) DO UPDATE SET
        username=excluded.username

    conn.commit()

def generate_game_code(length=6):
    return ''.join(random.choices(string.ascii_uppercase + string.digits, k=length))


def user_in_active_game(telegram_id: int):
    conn = get_connection()
    cursor = conn.cursor()

    cursor.execute("""
        SELECT * FROM games
        WHERE (player1_id = ? OR player2_id = ?)
        AND state != 'ENDED'
    """, (telegram_id, telegram_id))

    game = cursor.fetchone()
    conn.close()
    return game is not None


def create_game(player1_id: int):
    conn = get_connection()
    cursor = conn.cursor()

    code = generate_game_code()

    # Ensure uniqueness
    cursor.execute("SELECT 1 FROM games WHERE code = ?", (code,))
    while cursor.fetchone():
        code = generate_game_code()
        cursor.execute("SELECT 1 FROM games WHERE code = ?", (code,))

    cursor.execute("""
        INSERT INTO games (code, player1_id, current_turn, state)
        VALUES (?, ?, ?, ?)
    """, (code, player1_id, player1_id, "WAITING_FOR_PLAYER"))

    conn.commit()
    conn.close()

    return code

def join_game(code: str, player2_id: int):
    conn = get_connection()
    cursor = conn.cursor()

    cursor.execute("""
        SELECT player1_id, player2_id, state
        FROM games
        WHERE code = ?
    """, (code,))
    
    game = cursor.fetchone()

    if not game:
        conn.close()
        return "NOT_FOUND"

    player1_id, existing_player2, state = game

    if state != "WAITING_FOR_PLAYER":
        conn.close()
        return "NOT_AVAILABLE"

    if player1_id == player2_id:
        conn.close()
        return "CANNOT_JOIN_SELF"

    if existing_player2 is not None:
        conn.close()
        return "FULL"

    # Assign player2 and update state
    cursor.execute("""
        UPDATE games
        SET player2_id = ?, state = ?
        WHERE code = ?
    """, (player2_id, "CHOOSING_TYPE", code))

    conn.commit()
    conn.close()

    return "SUCCESS"

def end_game_by_user(telegram_id: int):
    conn = get_connection()
    cursor = conn.cursor()

    cursor.execute("""
        SELECT code, player1_id, player2_id
        FROM games
        WHERE (player1_id = ? OR player2_id = ?)
        AND state != 'ENDED'
    """, (telegram_id, telegram_id))

    game = cursor.fetchone()

    if not game:
        conn.close()
        return None

    code, player1_id, player2_id = game

    cursor.execute("""
        UPDATE games
        SET state = 'ENDED'
        WHERE code = ?
    """, (code,))

    conn.commit()
    conn.close()

    return {
        "code": code,
        "player1_id": player1_id,
        "player2_id": player2_id
    }

def update_game_prompt_options(code: str, selected_type: str, prompts: list):
    conn = get_connection()
    cursor = conn.cursor()

    # Store prompts as a single string separated by ||
    prompts_str = "||".join(prompts)

    cursor.execute("""
        UPDATE games
        SET selected_type = ?, selected_prompt = ?
        WHERE code = ?
    """, (selected_type, prompts_str, code))

    conn.commit()
    conn.close()

def get_game_prompt_options(code: str):
    conn = get_connection()
    cursor = conn.cursor()

    cursor.execute("""
        SELECT selected_prompt
        FROM games
        WHERE code = ?
    """, (code,))

    result = cursor.fetchone()
    conn.close()

    if not result or not result[0]:
        return []

    return result[0].split("||")
